20 def _build_query(geometry_geojson: dict, timeout: int = 60) -> str:
21 """
22 Build an Overpass QL query that fetches all OSM features (nodes, ways,
23 relations) within the given polygon boundary.
24 """
25 geom = shape(geometry_geojson)
26
27 if geom.geom_type == "MultiPolygon":
28 polygon = max(geom.geoms, key=lambda g: g.area)
29 else:
30 polygon = geom
31
32 simplified = polygon.simplify(0.0001, preserve_topology=True)
33 coords = list(simplified.exterior.coords)
34
35 poly_str = " ".join(f"{lat} {lon}" for lon, lat in coords)
36
37 query = f"""
38[out:json][timeout:{timeout}];
39(
40 nwr(poly:"{poly_str}");
41);
42out body;
43>;
44out skel qt;
45"""
46 return query

49 def _overpass_elements_to_geojson(data: dict) -> geojson.FeatureCollection:
50 """
51 Convert Overpass JSON response to a GeoJSON FeatureCollection.
52 Handles nodes, ways, and basic relations.
53 """
54 elements = data.get("elements", [])
55
56 nodes = {}
57 ways = {}
58 features = []
59
60 for elem in elements:
61 if elem["type"] == "node":
62 nodes[elem["id"]] = elem
63
64 for elem in elements:
65 if elem["type"] == "way":
66 ways[elem["id"]] = elem
67
68 for elem in elements:
69 feature = None
70 if elem["type"] == "node" and "tags" in elem:
71 feature = geojson.Feature(
72 geometry=geojson.Point([elem["lon"], elem["lat"]]),
73 properties={
74 "osm_id": elem["id"],
75 "osm_type": "node",
76 **(elem.get("tags", {})),
77 },
78 )
79 elif elem["type"] == "way" and "tags" in elem:
80 coords = []
81 for nd_ref in elem.get("nodes", []):
82 nd = nodes.get(nd_ref)
83 if nd:
84 coords.append([nd["lon"], nd["lat"]])
85
86 if len(coords) < 2:
87 continue
88
89 if len(coords) >= 4 and coords[0] == coords[-1]:
90 feature = geojson.Feature(
91 geometry=geojson.Polygon([coords]),
92 properties={
93 "osm_id": elem["id"],
94 "osm_type": "way",
95 **(elem.get("tags", {})),
96 },
97 )
98 else:
99 feature = geojson.Feature(
100 geometry=geojson.LineString(coords),
101 properties={
102 "osm_id": elem["id"],
103 "osm_type": "way",
104 **(elem.get("tags", {})),
105 },
106 )
107 elif elem["type"] == "relation" and "tags" in elem:
108 feature = geojson.Feature(
109 geometry=None,
110 properties={
111 "osm_id": elem["id"],
112 "osm_type": "relation",
113 **(elem.get("tags", {})),
114 },
115 )
116
117 if feature:
118 features.append(feature)
119
120 return geojson.FeatureCollection(features)
